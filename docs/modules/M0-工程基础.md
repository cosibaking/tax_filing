## 一、模块职责与边界

提供横切基础设施：项目脚手架、Prisma 全局 Schema、API 信封、认证框架、加密/脱敏、附件存储、通知、审计、Cron/BullMQ 框架。

### 3.2 全局 Prisma 模型

```prisma
model Member {
  id            BigInt   @id @default(autoincrement())
  phone         String   @unique @db.VarChar(20)
  passwordHash  String?  @map("password_hash")
  name          String?  @db.VarChar(64)
  email         String?  @db.VarChar(128)
  deleted       Boolean  @default(false)
  deletedAt     DateTime? @map("deleted_at")
  createdAt     DateTime @default(now()) @map("created_at")
  updatedAt     DateTime @updatedAt @map("updated_at")
  @@map("members")
}

model AdminUser {
  id           BigInt   @id @default(autoincrement())
  username     String   @unique
  passwordHash String   @map("password_hash")
  role         String   @default("advisor") // advisor, admin
  deleted      Boolean  @default(false)
  deletedAt    DateTime? @map("deleted_at")
  createdAt    DateTime @default(now()) @map("created_at")
  @@map("admin_users")
}

model OtpCode {
  id        BigInt   @id @default(autoincrement())
  phone     String   @db.VarChar(20)
  code      String   @db.VarChar(8)
  purpose   String   // login, register
  expiresAt DateTime @map("expires_at")
  used      Boolean  @default(false)
  createdAt DateTime @default(now()) @map("created_at")
  @@index([phone, purpose])
  @@map("otp_codes")
}

model SysAttachment {
  id         BigInt   @id @default(autoincrement())
  memberId   BigInt?  @map("member_id")
  fileName   String   @map("file_name")
  mimeType   String   @map("mime_type")
  size       Int
  storageKey String   @map("storage_key")
  deleted    Boolean  @default(false)
  deletedAt  DateTime? @map("deleted_at")
  createdAt  DateTime @default(now()) @map("created_at")
  @@map("sys_attachments")
}

model MemberNotice {
  id        BigInt   @id @default(autoincrement())
  memberId  BigInt   @map("member_id")
  type      String   // material_reminder, filing_done, statement
  title     String
  content   String   @db.Text
  read      Boolean  @default(false)
  deleted   Boolean  @default(false)
  createdAt DateTime @default(now()) @map("created_at")
  @@map("member_notices")
}

model ComplianceAuditLog {
  id           BigInt   @id @default(autoincrement())
  entityType   String   @map("entity_type")
  entityId     BigInt   @map("entity_id")
  action       String
  operatorId   BigInt   @map("operator_id")
  operatorType String   @map("operator_type") // member, admin
  before       Json?
  after        Json?
  ip           String?  @db.VarChar(45)
  createdAt    DateTime @default(now()) @map("created_at")
  @@index([entityType, entityId])
  @@map("compliance_audit_logs")
}
```
